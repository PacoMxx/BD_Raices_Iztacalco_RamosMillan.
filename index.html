¿Por qué nuestra colonia se llama Gabriel Ramos Millán? La historia del «Apóstol del Maíz» y las raíces de nuestras cuadras

Cada mañana caminamos por Sur 157 rumbo al Mercado Bramadero o cruzamos Oriente 100-A en Tlacotal, pero pocos vecinos jóvenes conocen el origen del nombre que nos da identidad. Gabriel Ramos Millán fue el abogado e impulsor de la Comisión Nacional del Maíz que modernizó el campo mexicano a finales de los años 40.

Tras su trágico fallecimiento el 26 de septiembre de 1949 en el accidente aéreo del Pico del Fraile (en el Popocatépetl, donde también viajaba la actriz Blanca Estela Pavón), los terrenos en expansión al sur del pueblo originario de Iztacalco fueron bautizados en su honor.

En los años 50 y 60, los primeros habitantes levantaron con esfuerzo propio las escuelas, las panaderías tradicionales y los mercados públicos (como el Mercado 204 Bramadero y el Mercado Tlacotal), transformando antiguos llanos de humedad en el corazón comercial y administrativo de nuestra alcaldía.

/**
 * BACKEND SERVERLESS BLINDADO: RAÍCES DE IZTACALCO (EDICIÓN GABRIEL RAMOS MILLÁN)
 * Candado 1: Sanitización Anti-XSS + Honeypot Anti-Bots + Estatus "Pendiente" obligatorio.
 * Candado 2: Filtrado estricto de filas ("Activo"/"Aprobado") y ocultamiento de columnas privadas.
 */

// 1. LISTA BLANCA DE COLUMNAS PÚBLICAS (Jamás envía Correo_Privado, Fecha_Cobro ni Notas_Cliente)
const COLUMNAS_PUBLICAS = {
  'Config_Alertas': ['Estado_Vial', 'Evento_Deportiva', 'Tianguis_Hoy', 'Aviso_Agua', 'Patrocinador_Alerta'],
  'Cronicas_Noticias': ['ID', 'Titulo', 'Categoria', 'Sector', 'Resumen', 'Contenido', 'Imagen_URL', 'Fecha'],
  'Directorio_Negocios': ['ID', 'Nombre', 'Giro', 'Sector', 'Calle', 'Anio_Fundacion', 'Plan', 'Cupon_Activo', 'Catalogo_Rapido', 'WhatsApp'],
  'Tianguis_Clasificados': ['ID', 'Producto', 'Precio', 'Vendedor', 'Punto_Entrega', 'Destacado', 'Foto_URL', 'WhatsApp'],
  'Comunidad_Servicios': ['ID', 'Tipo', 'Titulo', 'Calle_Zona', 'Descripcion', 'Destacado_Pagado', 'WhatsApp']
};

function doGet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const payload = {
    alertas: limpiarDatosPublicos(ss.getSheetByName('Config_Alertas'), 'Config_Alertas', false),
    cronicas: limpiarDatosPublicos(ss.getSheetByName('Cronicas_Noticias'), 'Cronicas_Noticias', true),
    negocios: limpiarDatosPublicos(ss.getSheetByName('Directorio_Negocios'), 'Directorio_Negocios', true),
    tianguis: limpiarDatosPublicos(ss.getSheetByName('Tianguis_Clasificados'), 'Tianguis_Clasificados', true),
    comunidad: limpiarDatosPublicos(ss.getSheetByName('Comunidad_Servicios'), 'Comunidad_Servicios', true)
  };
  return ContentService.createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents || '{}');

    // CANDADO 1.A: Si el bot llenó el campo trampa Honeypot, ignorar silenciosamente
    if (body.website_url_trap && String(body.website_url_trap).trim() !== '') {
      return ContentService.createTextOutput(JSON.stringify({ ok: true, blocked: 'honeypot' }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName('Solicitudes_Vecinales') || ss.insertSheet('Solicitudes_Vecinales');

    // CANDADO 1.B: Sanitizar texto y forzar estatus inicial a "Pendiente"
    sheet.appendRow([
      new Date(),
      sanitizarTexto(body.tipo),
      sanitizarTexto(body.sector),
      sanitizarTexto(body.titulo),
      sanitizarTexto(body.calle),
      sanitizarTelefono(body.whatsapp),
      sanitizarTexto(body.precio),
      sanitizarTexto(body.descripcion),
      'Pendiente' // Nada se publica en vivo hasta que tú cambies esta celda desde tu celular
    ]);

    return ContentService.createTextOutput(JSON.stringify({ ok: true, status: 'Pendiente' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// CANDADO 2: Filtra filas no aprobadas y elimina columnas sensibles internas
function limpiarDatosPublicos(sheet, nombreHoja, requiereAprobacion) {
  if (!sheet) return [];
  const rows = sheet.getDataRange().getValues();
  const headers = rows.shift();
  const permitidas = COLUMNAS_PUBLICAS[nombreHoja] || headers;

  return rows
    .filter(r => {
      if (!requiereAprobacion) return true;
      const idxEstatus = headers.indexOf('Estatus');
      const idxPago = headers.indexOf('Estatus_Pago');
      const valorEstatus = idxEstatus > -1 ? String(r[idxEstatus]).trim() : '';
      const valorPago = idxPago > -1 ? String(r[idxPago]).trim() : '';
      return valorEstatus === 'Aprobado' || valorPago === 'Activo';
    })
    .map(r => {
      const obj = {};
      headers.forEach((h, i) => {
        if (permitidas.indexOf(h) > -1) {
          obj[h] = r[i];
        }
      });
      return obj;
    });
}

function sanitizarTexto(valor) {
  return String(valor || '')
    .replace(/[<>]/g, '')
    .replace(/^[=+\-@]/, "'") // Previene inyección de fórmulas en Google Sheets
    .trim()
    .slice(0, 500);
}

function sanitizarTelefono(tel) {
  return String(tel || '').replace(/\D/g, '').slice(-10);
}
